```

│       │   └── ...

│           └── ...
```

**Interface do Plugin:**
```php

// src/Core/Plugin/PluginInterface.php

namespace DomainSystem\Core\Plugin;

interface PluginInterface
{
    /**
     * Registra os hooks, rotas e serviços do plugin.
     * Chamado durante o boot do Kernel.
     */
    public function register(): void;

    /**
     * Retorna o nome do plugin (para identificação).
     */
    public function getName(): string;

    /**
     * Retorna a versão do plugin.
     */
    public function getVersion(): string;

    /**
     * Retorna as dependências (nomes de outros plugins) que este plugin precisa.
     */
    public function getDependencies(): array;

    /**
     * Define se o plugin está ativo.
     */
    public function isActive(): bool;
}
```

### 2. **Sistema de Rotas (Router)**

Os plugins precisam registrar suas próprias rotas.

**Estrutura Proposta:**

```php
// src/Core/Routing/Router.php

namespace DomainSystem\Core\Routing;

class Router
{
    private array $routes = [];

    public function addRoute(string $method, string $path, callable|array $handler, string $plugin = ''): void
    {
        $this->routes[$method][$path] = [
            'handler' => $handler,
            'plugin' => $plugin
        ];
    }

    public function dispatch(string $method, string $uri): mixed
    {
        // Remove query string
        $uri = strtok($uri, '?');

        // Busca rota exata
        if (isset($this->routes[$method][$uri])) {
            return $this->executeHandler($this->routes[$method][$uri]['handler']);
        }

        // Busca rota com parâmetros (ex: /pacientes/{id})
        foreach ($this->routes[$method] as $route => $config) {
            $pattern = preg_replace('/\{[a-zA-Z_]+\}/', '([^/]+)', $route);
            if (preg_match('#^' . $pattern . '$#', $uri, $matches)) {
                array_shift($matches);
                return $this->executeHandler($config['handler'], $matches);
            }
        }

        throw new \Exception("Rota não encontrada: $uri", 404);
    }

    private function executeHandler(callable|array $handler, array $params = []): mixed
    {
        if (is_callable($handler)) {
            return call_user_func_array($handler, $params);
        }

        if (is_array($handler) && count($handler) === 2) {
            [$class, $method] = $handler;
            $instance = $this->container->make($class);
            return call_user_func_array([$instance, $method], $params);
        }

        throw new \Exception("Handler inválido.");
    }
}
```

### 3. **Abstração de Banco de Dados (Database Plugin)**

O plugin de banco de dados deve ser o primeiro a ser carregado, pois será dependência de todos os outros.

**Estrutura Proposta:**

```php
// src/Plugins/Database/Plugin.php

namespace DomainSystem\Plugins\Database;

use DomainSystem\Core\Plugin\PluginInterface;
use DomainSystem\Core\Container\Container;

class Plugin implements PluginInterface
{
    public function getName(): string { return 'database'; }
    public function getVersion(): string { return '1.0.0'; }
    public function getDependencies(): array { return []; }
    public function isActive(): bool { return true; }

    public function register(): void
    {
        // Registra o serviço de conexão no Container
        $container = Container::getInstance();
        
        $container->singleton(Connection::class, function($c) {
            return new Connection(
                getenv('DB_HOST'),
                getenv('DB_NAME'),
                getenv('DB_USER'),
                getenv('DB_PASS')
            );
        });

        $container->singleton(QueryBuilder::class, function($c) {
            return new QueryBuilder($c->make(Connection::class));
        });

        // Adiciona hooks para outros plugins
        add_action('database.connected', function() {
            // Disparado quando o banco é conectado
        });
    }
}
```

**Abstração do QueryBuilder (Motor de Consultas):**

```php
// src/Plugins/Database/QueryBuilder.php

namespace DomainSystem\Plugins\Database;

class QueryBuilder
{
    private Connection $connection;

    public function __construct(Connection $connection)
    {
        $this->connection = $connection;
    }

    public function table(string $table): QueryBuilder
    {
        $this->table = $table;
        return $this;
    }

    public function select(array $columns = ['*']): QueryBuilder
    {
        $this->columns = $columns;
        return $this;
    }

    public function where(string $column, string $operator, mixed $value): QueryBuilder
    {
        $this->wheres[] = [$column, $operator, $value];
        return $this;
    }

    public function get(): array
    {
        $sql = $this->buildSelect();
        $stmt = $this->connection->getPdo()->prepare($sql);
        $stmt->execute($this->bindings);
        return $stmt->fetchAll(\PDO::FETCH_ASSOC);
    }

    public function first(): ?array
    {
        $sql = $this->buildSelect() . ' LIMIT 1';
        $stmt = $this->connection->getPdo()->prepare($sql);
        $stmt->execute($this->bindings);
        return $stmt->fetch(\PDO::FETCH_ASSOC) ?: null;
    }

    public function insert(array $data): int
    {
        $columns = array_keys($data);
        $placeholders = array_map(fn($col) => ":$col", $columns);
        $sql = "INSERT INTO {$this->table} (" . implode(', ', $columns) . ") VALUES (" . implode(', ', $placeholders) . ")";
        $stmt = $this->connection->getPdo()->prepare($sql);
        $stmt->execute($data);
        return (int) $this->connection->getPdo()->lastInsertId();
    }

    public function update(array $data): int
    {
        $sets = [];
        foreach ($data as $column => $value) {
            $sets[] = "$column = :$column";
        }
        $sql = "UPDATE {$this->table} SET " . implode(', ', $sets) . " WHERE 1=1";
        // Adicionar wheres...
        $stmt = $this->connection->getPdo()->prepare($sql);
        $stmt->execute(array_merge($data, $this->bindings));
        return $stmt->rowCount();
    }

    // ... métodos delete, join, orderBy, groupBy, etc.
}
```

### 4. **Sistema de Templates (UI Plugin)**

O plugin de UI será responsável por renderizar as views (HTML) e gerenciar temas.

**Estrutura Proposta:**

```php
// src/Plugins/UI/Plugin.php

namespace DomainSystem\Plugins\UI;

use DomainSystem\Core\Plugin\PluginInterface;

class Plugin implements PluginInterface
{
    public function getName(): string { return 'ui'; }
    public function getVersion(): string { return '1.0.0'; }
    public function getDependencies(): array { return ['database']; }
    public function isActive(): bool { return true; }

    public function register(): void
    {
        $container = Container::getInstance();
        
        $container->singleton(TemplateEngine::class, function($c) {
            return new TemplateEngine(
                $c->make(ViewFinder::class)
            );
        });

        // Registra um hook para renderizar a página
        add_action('kernel.render', function($router) {
            $templateEngine = Container::getInstance()->make(TemplateEngine::class);
            echo $templateEngine->render('layout', [
                'content' => $router->getCurrentContent()
            ]);
        });
    }
}
```

**Motor de Templates (SSR):**

```php
// src/Plugins/UI/TemplateEngine.php

namespace DomainSystem\Plugins\UI;

class TemplateEngine
{
    private ViewFinder $finder;
    private array $globals = [];

    public function __construct(ViewFinder $finder)
    {
        $this->finder = $finder;
    }

    public function render(string $template, array $data = []): string
    {
        $data = array_merge($this->globals, $data);
        extract($data);

        $templatePath = $this->finder->find($template);

        ob_start();
        include $templatePath;
        return ob_get_clean();
    }

    public function addGlobal(string $key, mixed $value): void
    {
        $this->globals[$key] = $value;
    }
}
```

### 5. **Sistema de Views (Temas)**

Os plugins podem fornecer suas próprias views.

**Estrutura Proposta:**

```php
// src/Plugins/UI/ViewFinder.php

namespace DomainSystem\Plugins\UI;

class ViewFinder
{
    private array $paths = [];

    public function addPath(string $path, string $namespace = ''): void
    {
        if ($namespace) {
            $this->paths[$namespace] = $path;
        } else {
            $this->paths[] = $path;
        }
    }

    public function find(string $template): string
    {
        // Se tiver namespace (ex: "dashboard::index")
        if (strpos($template, '::') !== false) {
            [$namespace, $name] = explode('::', $template);
            if (isset($this->paths[$namespace])) {
                return $this->paths[$namespace] . '/' . str_replace('.', '/', $name) . '.php';
            }
        }

        // Busca nos caminhos padrão
        foreach ($this->paths as $path) {
            $file = $path . '/' . str_replace('.', '/', $template) . '.php';
            if (file_exists($file)) {
                return $file;
            }
        }

        throw new \Exception("Template '$template' não encontrado.");
    }
}
```

### 6. **Exemplo de Plugin de Negócio (Pacientes)**

Agora, um plugin de negócio (como Pacientes) pode ser construído de forma completamente desacoplada.

```php
// src/Plugins/Pacientes/Plugin.php

namespace DomainSystem\Plugins\Pacientes;

use DomainSystem\Core\Plugin\PluginInterface;

class Plugin implements PluginInterface
{
    public function getName(): string { return 'pacientes'; }
    public function getVersion(): string { return '1.0.0'; }
    public function getDependencies(): array { return ['database', 'ui', 'routing']; }
    public function isActive(): bool { return true; }

    public function register(): void
    {
        $container = Container::getInstance();
        
        // Registra o repositório
        $container->singleton(PacienteRepository::class, function($c) {
            return new PacienteRepository(
                $c->make(QueryBuilder::class)
            );
        });

        // Registra o controller
        $container->singleton(PacienteController::class, function($c) {
            return new PacienteController(
                $c->make(PacienteRepository::class)
            );
        });

        // Registra as rotas
        add_action('router.register', function($router) {
            $router->addRoute('GET', '/pacientes', [PacienteController::class, 'list'], 'pacientes');
            $router->addRoute('POST', '/pacientes', [PacienteController::class, 'create'], 'pacientes');
            $router->addRoute('GET', '/pacientes/{id}', [PacienteController::class, 'show'], 'pacientes');
            $router->addRoute('PUT', '/pacientes/{id}', [PacienteController::class, 'update'], 'pacientes');
            $router->addRoute('DELETE', '/pacientes/{id}', [PacienteController::class, 'delete'], 'pacientes');
        });

        // Registra hooks para modificar o HTML
        add_filter('ui.menu_items', function($items) {
            $items['pacientes'] = [
                'label' => 'Pacientes',
                'icon' => 'fa-users',
                'url' => '/pacientes'
            ];
            return $items;
        });
    }
}
```
